+const DEFAULT_SYMBOLS = ["BTC/USD", "ETH/USD", "SOL/USD", "AVAX/USD", "DOGE/USD", "XRP/USD"];
+
+export async function GET(request: Request) {
+  try {
+    const url = new URL(request.url);
+    const raw = url.searchParams.get("symbols");
+    const symbols = raw ? raw.split(",").map(s => s.trim()) : DEFAULT_SYMBOLS;
+
+    const syms = symbols.map(s => encodeURIComponent(s)).join(",");
+
+    const res = await fetch(
+      `https://data.alpaca.markets/v1beta3/crypto/us/latest/bars?symbols=${syms}`,
+      {
+        headers: {
+          "APCA-API-KEY-ID": ALPACA_API_KEY,
+          "APCA-API-SECRET-KEY": ALPACA_SECRET,
+        },
+      }
+    );
+
+    if (!res.ok) {
+      const text = await res.text();
+      throw new Error(`Alpaca ${res.status}: ${text}`);
+    }
+
+    const data = await res.json();
+    const bars = data.bars ?? {};
+
+    const quotes = Object.entries(bars).map(([symbol, bar]: [string, any]) => ({
+      symbol,
+      price: bar.c,
+      open: bar.o,
+      high: bar.h,
+      low: bar.l,
+      volume: bar.v,
+      timestamp: bar.t,
+    }));
+
+    return NextResponse.json({ ok: true, quotes, count: quotes.length, source: "alpaca-paper" });
+  } catch (error) {
+    const message = error instanceof Error ? error.message : "Alpaca crypto failed";
+    return NextResponse.json({ ok: false, error: message, quotes: [] }, { status: 503 });

+/**
+ * Alpaca Crypto Historical Data
+ * Provides OHLCV bars for technical analysis (EMA, RSI, change1d, etc.)
+ */
+import "server-only";
+
+const ALPACA_API_KEY = process.env.ALPACA_API_KEY ?? "";
+const ALPACA_SECRET = process.env.ALPACA_SECRET ?? "";
+const DATA_URL = "https://data.alpaca.markets/v1beta3/crypto/us";
+const TIMEOUT_MS = 10_000;
+
+export type AlpacaBar = {
+  t: string;  // timestamp
+  o: number;  // open
+  h: number;  // high
+  l: number;  // low
+  c: number;  // close
+  v: number;  // volume
+  vw: number; // vwap
+};

+export type AlpacaCryptoHistory = {
+  symbol: string;
+  bars: AlpacaBar[];
+  change1d: number;
+  change1dPct: number;
+  high52w: number;
+  low52w: number;
+  vol20: number;
+  ema9: number | null;
+  ema21: number | null;
+  ema50: number | null;
+  rsi14: number | null;
+  vwap: number | null;

+
+async function alpacaFetch<T>(url: string): Promise<T> {
+  const controller = new AbortController();
+  const timer = setTimeout(() => controller.abort(), TIMEOUT_MS);
+  try {
+    const res = await fetch(url, {
+      headers: {
+        "APCA-API-KEY-ID": ALPACA_API_KEY,
+        "APCA-API-SECRET-KEY": ALPACA_SECRET,
+      },
+      signal: controller.signal,
+    });
+    if (!res.ok) {
+      const text = await res.text().catch(() => res.statusText);
+      throw new Error(`Alpaca ${res.status}: ${text}`);
+    }
+    return res.json() as Promise<T>;
+  } finally {
+    clearTimeout(timer);
+  }
+}
+
+function calcEMA(closes: number[], period: number): number | null {
+  if (closes.length < period) return null;
+  const k = 2 / (period + 1);
+  let ema = closes.slice(0, period).reduce((a, b) => a + b, 0) / period;
+  for (let i = period; i < closes.length; i++) {
+    ema = closes[i]! * k + ema * (1 - k);
+  }
+  return ema;
+}
+
+function calcRSI(closes: number[], period = 14): number | null {
+  if (closes.length < period + 1) return null;
+  let gains = 0, losses = 0;
+  for (let i = closes.length - period; i < closes.length; i++) {
+    const diff = closes[i]! - closes[i - 1]!;
+    if (diff > 0) gains += diff;
+    else losses -= diff;
+  }
+  if (losses === 0) return 100;
+  const rs = gains / losses;
+  return 100 - 100 / (1 + rs);
+}

+/**
+ * Fetch daily bars for a crypto ticker and compute technical indicators.
+ */
+export async function getAlpacaCryptoHistory(ticker: string, limit = 60): Promise<AlpacaCryptoHistory> {
+  const symbol = toAlpacaSymbol(ticker);
+  const encoded = encodeURIComponent(symbol);
+
+  const data = await alpacaFetch<{
+    bars: Record<string, AlpacaBar[]>;
+  }>(`${DATA_URL}/bars?symbols=${encoded}&timeframe=1H&limit=${limit}&sort=asc`);
+
+  const bars = data.bars?.[symbol] ?? [];
+
+  if (bars.length < 2) {
+    throw new Error(`Sin datos históricos suficientes para ${symbol}`);
+  }
+
+  const closes = bars.map(b => b.c);
+  const today = bars[bars.length - 1]!;
+  const yesterday = bars[bars.length - 2]!;
+
+  const change1d = today.c - yesterday.c;
+  const change1dPct = (change1d / yesterday.c) * 100;
+  const high52w = Math.max(...bars.map(b => b.h));
+  const low52w = Math.min(...bars.map(b => b.l));
+  const vol20 = bars.slice(-20).reduce((s, b) => s + b.v, 0) / Math.min(20, bars.length);
+  const ema9 = calcEMA(closes, 9);
+  const ema21 = calcEMA(closes, 21);
+  const ema50 = calcEMA(closes, 50);
+  const rsi14 = calcRSI(closes);
+  const vwap = today.vw;
+
+  return {
+    symbol,
+    bars,
+    change1d,
+    change1dPct,
+    high52w,
+    low52w,
+    vol20,
+    ema9,
+    ema21,
+    ema50,
+    rsi14,
+    vwap,
+  };
+}
