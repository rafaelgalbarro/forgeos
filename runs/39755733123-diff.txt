+const DEFAULT_SYMBOLS = ["BTC/USD", "ETH/USD", "SOL/USD", "AVAX/USD", "DOGE/USD", "XRP/USD"];
+
+export async function GET(request: Request) {
+  try {
+    const url = new URL(request.url);
+    const raw = url.searchParams.get("symbols");
+    const symbols = raw ? raw.split(",").map(s => s.trim()) : DEFAULT_SYMBOLS;
+
+    const syms = symbols.map(s => encodeURIComponent(s)).join(",");
+
+    const res = await fetch(
+      `https://data.alpaca.markets/v1beta3/crypto/us/latest/bars?symbols=${syms}`,
+      {
+        headers: {
+          "APCA-API-KEY-ID": ALPACA_API_KEY,
+          "APCA-API-SECRET-KEY": ALPACA_SECRET,
+        },
+      }
+    );
+
+    if (!res.ok) {
+      const text = await res.text();
+      throw new Error(`Alpaca ${res.status}: ${text}`);
+    }
+
+    const data = await res.json();
+    const bars = data.bars ?? {};
+
+    const quotes = Object.entries(bars).map(([symbol, bar]: [string, any]) => ({
+      symbol,
+      price: bar.c,
+      open: bar.o,
+      high: bar.h,
+      low: bar.l,
+      volume: bar.v,
+      timestamp: bar.t,
+    }));
+
+    return NextResponse.json({ ok: true, quotes, count: quotes.length, source: "alpaca-paper" });
+  } catch (error) {
+    const message = error instanceof Error ? error.message : "Alpaca crypto failed";
+    return NextResponse.json({ ok: false, error: message, quotes: [] }, { status: 503 });
+  }

+function alpacaHeaders() {
+  return {
+    "APCA-API-KEY-ID": ALPACA_API_KEY,

+async function alpacaFetch<T>(url: string, init?: RequestInit): Promise<T> {
+  const controller = new AbortController();
+  const timer = setTimeout(() => controller.abort(), TIMEOUT_MS);
+  try {
+    const res = await fetch(url, {
+      ...init,
+      headers: { ...alpacaHeaders(), ...(init?.headers ?? {}) },
+      signal: controller.signal,
+    });
+    if (!res.ok) {
+      const text = await res.text().catch(() => res.statusText);
+      throw new Error(`Alpaca ${res.status}: ${text}`);
+    }
+    return res.json() as Promise<T>;
+  } finally {
+    clearTimeout(timer);
+  }

+
+  // Para crypto Alpaca usamos qty fraccional o notional
+  const orderBody: Record<string, unknown> = {
+    symbol: alpacaSymbol,
+    side,
+    type: params.limitPrice ? "limit" : "market",
+    time_in_force: "gtc",
+  };
+
+  // Usamos qty directa (shares = unidades de crypto)
+  if (params.quantity > 0) {
+    orderBody.qty = String(params.quantity);
+  }

+    orderBody.limit_price = String(params.limitPrice.toFixed(2));

+  }>(`${ALPACA_ENDPOINT}/orders`, {

+  });
+
+  console.log(
+    `[AlpacaSubmit] ✅ Orden enviada: id=${order.id} symbol=${order.symbol} ` +
+    `status=${order.status} qty=${order.qty}`

+
+  return {
+    orderId: order.id,
+    clientOrderId: order.client_order_id,
+    symbol: order.symbol,
+    status: order.status,
